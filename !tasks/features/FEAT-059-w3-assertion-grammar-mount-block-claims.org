* @then block-claim grammar (authored in %capture_type, not Rust)
#+BEGIN_SRC
%capture_type claim {
    ( "not" $negated:claim )
  | ( $prop:claim_prop $op:claim_op $value:expr )
  | ( $prop:claim_prop )                       // bare = truthy/exists
}
%capture_type claim_prop {
    ( "computed" "(" $css:ident ")" )          // forces layout rung (W1 inference)
  | ( "$" $signal:ident )
  | ( $name:ident )                            // state|text|value|count|attr|…
}
%capture_type claim_op { "==" | "!=" | ">=" | "<=" | "<" | ">" | "~>" | "~=" | "matches" }
%capture_type claim_block { ( $claim:claim ";"? )* }

%macro then { %form { @then $subject:selector { $claims:claim_block } }
              %emit js { /* each claim → throwing check; computed() ⇒ rung=layout */ } }
#+END_SRC
