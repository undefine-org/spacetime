** Rationale
Pretext is a ~15KB pure-TS library that provides canvas-based text measurement with 7680/7680 cross-browser accuracy. By vendoring it into the runtime, any %primitive can use =ST.text= for layout-reflow-free text measurement.

Wrap the raw IIFE in an ST.text namespace integration:
#+begin_src javascript
// public/runtime/pretext.js
// Pretext — canvas-based text measurement (vendored from @chenglou/pretext)
(function() {
  // ... raw IIFE content ...

  // Expose via ST namespace
  const P = __PretextInternal;
  ST.text = {
    prepare: P.prepare,
    layout: P.layout,
    prepareWithSegments: P.prepareWithSegments,
    layoutWithLines: P.layoutWithLines,
    clearCache: P.clearCache,
    setLocale: P.setLocale,
    // Thin wrapper for single-segment width measurement via canvas
    measureWidth: function(text, font) {
      // Uses pretext's internal canvas context for consistency
      var ctx = this._ctx;
      if (!ctx) {
        if (typeof OffscreenCanvas !== 'undefined') {
          ctx = new OffscreenCanvas(1, 1).getContext('2d');
        } else if (typeof document !== 'undefined') {
          ctx = document.createElement('canvas').getContext('2d');
        }
        this._ctx = ctx;
      }
      if (!ctx) return 0;
      ctx.font = font;
      return ctx.measureText(text).width;
    },
    _ctx: null
  };
})();
#+end_src
