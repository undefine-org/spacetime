* Design (confirmed with user)
A %migration becomes a SELF-CONTAINED CAPSULE: it embeds the retired %macro definitions VERBATIM (grammar + %binds semantics) alongside the migration metadata, so stdlib proper only ever carries current syntax. ONE migration may retire SEVERAL macros (a wave is one retirement, not one file per form).

#+BEGIN_SRC
%migration reactive-surface {
  %date 2026-06-09
  %docs "FEAT-072 wave: the paren-macro reactive surface retired for `:` / `<-`."

  // retired definitions, verbatim (from stdlib/macros/data.st @ 75ce7975^)
  %macro bind { %form { @bind(text: $text:expr? = undefined, …) } %if $text { %binds { bind-text(…) } } … }
  %macro show { %creates @show %form { @show(when: $condition:expr, mode: $mode:string = "display") } %binds { bind-visible(&self, condition: $condition, mode: $mode) } }
  %macro input-bind { %creates @input %form { … } %binds { bind-input(&self, value: $value, event: $event, debounce: $debounce) -> { $value } } }

  // mechanical rewrites (per-shape data; pattern reuses the %form PEG)
  %rewrite { @bind(text: $x:expr) } => { text <- `$x`; }
  %rewrite { @bind(attr: $n:string, value: $v:expr) } => { `$n` <- `$v`; }
  %rewrite { @bind(class: $c:string, when: $cond:expr) } => { .`$c`: `$cond`; }
  %rewrite { @bind(style: $p:string, value: $v:expr) } => { `$p`: `$v`; }
  %rewrite { @bind(visible: $v:expr) } => { .shown: `$v`; }

  // manual guidance, keyed by directive
  %hint for @show "Replace with a reactive class toggle — `.hidden: !$condition;` + a `.hidden { display: none }` rule."
  %hint for @input "Replace with `value <- $s;` + `@on input { $s <- &el.value; }`."
}
#+END_SRC
